import { Suspense } from 'react'
import { connection } from 'next/server'
import AppFrame from '@/components/AppFrame'
import { getAssetWorkspace } from './actions'
import { AssetWorkspace } from './workspace'
import './asset.css'


export function AssetScreen({ section = 'assets', tab = 'overview' }: { section?: 'assets' | 'debts' | 'history'; tab?: string }) {
  return <Suspense fallback={<p className="p-8" role="status">กำลังโหลดสินทรัพย์…</p>}><Content section={section} tab={tab} /></Suspense>
}

async function Content({ section, tab }: { section: 'assets' | 'debts' | 'history'; tab: string }) {
  await connection()
  const { data, categories, canWrite, ledger, history } = await getAssetWorkspace()
  return <AppFrame><AssetWorkspace initial={data} categories={categories} canWrite={canWrite} ledger={ledger} history={history} initialSection={section} initialTab={tab} /></AppFrame>
}
