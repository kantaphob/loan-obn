function displayDate(value: string) { return value ? new Date(`${value}T00:00:00+07:00`).toLocaleDateString('th-TH', { timeZone: 'Asia/Bangkok' }) : 'ยังไม่ระบุ' }
function warrantyEnd(start: string, months: number) {
  if (!start || !months) return ''
  const d = new Date(`${start}T00:00:00Z`); const day = d.getUTCDate()
  d.setUTCDate(1); d.setUTCMonth(d.getUTCMonth() + months)
  const last = new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth() + 1, 0)).getUTCDate()
  d.setUTCDate(Math.min(day, last)); return d.toISOString().slice(0, 10)
}

        {round ? <><div className="asset-summary-row"><span>ยอดต้นรอบ {baht(round.opening)} · MRR อ้างอิง {round.mrr}%<small>ค่าธรรมเนียม {baht(round.fee)} · {round.notes}</small></span><button className="asset-button secondary" disabled={!editable || loan.payments.some(p => p.roundId === round.id) || round.id !== latest?.id} onClick={() => { if (!leaveForm()) return; closeEditors(); setRoundEditor(round) }}>ตั้งค่าสัญญา</button></div>{!round.effectiveOn ? <p className="asset-notice">กรุณาตั้งวันที่เริ่มสัญญาก่อนบันทึกชำระหรือนำเข้า CSV</p> : null}<p className="asset-hint">ดอกเบี้ยรวมอ้างอิงจากสัญญา: {round.referenceInterest ? baht(round.referenceInterest) : 'ไม่ระบุ'} · เก็บแยกจากดอกเบี้ยจ่ายจริงและยอดจำลอง</p><p className="asset-hint">รอบที่มีประวัติชำระแล้วจะเก็บเงื่อนไขเดิมไว้ การเปลี่ยนอัตราให้เพิ่มรอบใหม่ · อัตราหลังปีที่ 7 ตั้งไว้เฉพาะค่าที่ให้มา สามารถปรับช่วงตามเอกสารสัญญาก่อนเริ่มใช้งาน</p><div className="asset-table-scroll"><table><thead><tr><th>เดือนในรอบ</th><th>อัตราต่อปี</th><th>ค่างวด</th></tr></thead><tbody>{round.periods.map(p => <tr key={p.from}><td>{p.from}–{p.to}</td><td>{p.floating ? `MRR − ${p.discount}% = ${Math.max(0, round.mrr - p.discount).toFixed(3)}%` : `${p.rate}%`}</td><td>{money(p.installment)}</td></tr>)}</tbody></table></div></> : <Empty>เพิ่มสัญญาเพื่อเริ่มติดตามการผ่อน</Empty>}
      </Panel> : null}
      {roundEditor ? <RoundEditor key={typeof roundEditor === 'string' ? `new-${home.id}` : roundEditor.id} initial={typeof roundEditor === 'string' ? undefined : roundEditor} latest={latest} remaining={latest ? balance(loan, latest) : 0} disabled={!editable} onCancel={closeEditors} onSave={async r => { const ok = await updateLoan({ ...loan, rounds: roundEditor === 'new' ? [...loan.rounds, r] : loan.rounds.map(x => x.id === r.id ? r : x) }); if (ok) { setRoundId(r.id); setRoundEditor(null) } }} /> : null}
      {round ? <Panel title="ประวัติชำระจริง" action={<div className="asset-tools"><label className="asset-button secondary">นำเข้า GHB.csv<input type="file" accept=".csv,text/csv" className="sr-only" disabled={!editable || !round.effectiveOn} onChange={e => { void importFile(e.target.files?.[0]); e.target.value = '' }} /></label><Add enabled={editable} disabled={!round.effectiveOn} onClick={() => open('payment')}>บันทึกชำระ</Add></div>}>
        <label className="asset-filter">รอบสัญญาที่แสดง <select value={round.id} onChange={e => { if (!leaveForm()) return; closeEditors(); setRoundId(e.target.value) }}>{loan.rounds.map(r => <option key={r.id} value={r.id}>{r.name}</option>)}</select></label>
        {!round.effectiveOn ? <p className="asset-notice">ยังไม่ได้ระบุวันที่เริ่มสัญญา จึงยังบันทึกชำระไม่ได้ <button type="button" className="asset-button secondary" onClick={() => { if (!leaveForm()) return; closeEditors(); setTab('overview') }}>ไปตั้งค่าสัญญา</button></p> : null}
        <p className="asset-hint">กำลังดู: {round.name} · รวมโปะไว้ในยอดชำระ ใช้ยอดดอกเบี้ยจริงจากธนาคาร · ยังไม่สร้างรายการรับ–จ่ายอัตโนมัติ</p>
        {importRows ? <form onSubmit={async e => { e.preventDefault(); const next = importRows.map(p => ({ ...p, id: crypto.randomUUID(), roundId: round.id })); const duplicates = next.some(p => loan.payments.some(old => old.roundId === p.roundId && old.reference === p.reference)); if (duplicates) { setError('มีเลขงวดจากไฟล์นี้อยู่แล้ว กรุณาตรวจสอบประวัติก่อนนำเข้าซ้ำ'); return } if (await updateLoan({ ...loan, payments: [...loan.payments, ...next] })) setImportRows(null) }}><h3>ตรวจสอบก่อนนำเข้า {importRows.length} รายการ</h3><div className="asset-table-scroll"><table><thead><tr><th>วันที่ชำระ (ต้องระบุ)</th><th>รายการ</th><th>ยอดชำระ</th><th>ดอกเบี้ย</th><th>เงินต้น</th></tr></thead><tbody>{importRows.map((p, i) => <tr key={i}><td><input aria-label={`วันที่ชำระรายการ ${i + 1}`} type="date" required value={/^\d{4}-\d{2}-\d{2}$/.test(p.date) ? p.date : ''} onChange={e => setImportRows(importRows.map((r, j) => j === i ? { ...r, date: e.target.value } : r))} /></td><td>{p.reference}</td><td>{money(p.amount)}</td><td>{money(p.interest)}</td><td>{money(p.principal)}</td></tr>)}</tbody></table></div><div className="asset-form-actions"><button type="button" className="asset-button secondary" onClick={closeEditors}>ยกเลิก</button><button className="asset-button" disabled={!editable}>ยืนยันนำเข้า</button></div></form> : null}
        <div className="asset-table-scroll"><table><thead><tr><th>วันที่ / อ้างอิง</th><th>ยอดชำระ</th><th>ดอกเบี้ย</th><th>ตัดเงินต้น</th><th>เงินต้นคงเหลือ</th><th><span className="sr-only">จัดการ</span></th></tr></thead><tbody>{loan.payments.filter(p => p.roundId === round.id).toSorted((a, b) => b.date.localeCompare(a.date)).map(p => <tr key={p.id}><td>{displayDate(p.date)}<small>{p.reference}</small></td><td>{money(p.amount)}</td><td>{money(p.interest)}</td><td>{money(p.principal)}</td><td>{money(paidBalances.get(p.id) ?? 0)}</td><td>{renderRowActions('payment', p.id)}</td></tr>)}</tbody></table></div>{!loan.payments.some(p => p.roundId === round.id) ? <Empty>ยังไม่มีรายการชำระจริงในรอบนี้</Empty> : null}
      </Panel> : null}
      {tab !== 'payments' && round && simulation && baseSimulation ? <Panel title="จำลองแผนผ่อนของรอบสัญญา"><p className="asset-hint">จำลองตั้งแต่ต้นรอบ ด้วยดอกเบี้ยรายเดือน = เงินต้น × อัตราต่อปี ÷ 12 และถือ MRR คงที่ ใช้เปรียบเทียบแผน ไม่ใช่ยอดธนาคารหรือประมาณการจากยอดชำระปัจจุบัน</p><label className="asset-filter">โปะเพิ่มต่อเดือน (บาท) <input type="number" min="0" max="1000000000" step="100" value={extra} onChange={e => { const n = Number(e.target.value); setExtra(Number.isFinite(n) ? Math.max(0, Math.min(n, 1e9)) : 0) }} /></label><div className="asset-stats"><Stat label="ดอกเบี้ยจำลอง" value={baht(simulation.totalInterest)} /><Stat label="ระยะเวลาจำลอง" value={`${simulation.rows.length} เดือน`} note={simulation.remaining > 0 ? `ยังเหลือ ${baht(simulation.remaining)} เมื่อจบแผน` : 'ชำระหมดตามแผนจำลอง'} /><Stat label="ดอกเบี้ยต่างจากไม่โปะ" value={simulation.remaining === 0 && baseSimulation.remaining === 0 ? baht(baseSimulation.totalInterest - simulation.totalInterest) : 'ยังเทียบยอดรวมไม่ได้'} /></div><details><summary>ดูตารางผ่อนจำลอง</summary><div className="asset-table-scroll asset-forecast"><table><thead><tr><th>เดือน</th><th>ต้นงวด</th><th>อัตรา %</th><th>ดอกเบี้ย</th><th>จ่ายรวม</th><th>เงินต้น</th><th>คงเหลือ</th></tr></thead><tbody>{simulation.rows.map(r => <tr key={r.month}><td>{r.month}</td><td>{money(r.opening)}</td><td>{r.rate.toFixed(3)}</td><td>{money(r.interest)}</td><td>{money(r.payment)}</td><td>{money(r.principal)}</td><td>{money(r.remaining)}</td></tr>)}</tbody></table></div></details></Panel> : null}
    </> : null}
    <p className="asset-footnote">บันทึกในฐานข้อมูลและระบบสำรองเดิม · ราคาสินทรัพย์กับยอดหนี้แยกจากกัน</p>
